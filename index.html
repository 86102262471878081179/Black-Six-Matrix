System – Block

Felder: input_hash, output_hash, prev_block_hash, block_hash, ok
block_hash = H(input_hash || output_hash || prev_block_hash)

System – Chain

nur append() darf die Chain verändern
prev_block_hash jedes Blocks muss last_hash() des Vorgängers sein
validate_chain() prüft alle Blöcke + Verkettung

Consensus – Commit

use sha2::{Sha256, Digest};

pub fn hash_bytes(data: &[u8]) -> [u8; 32] {
    let mut hasher = Sha256::new();
    hasher.update(data);
    hasher.finalize().into()
}

System – block.rs

use serde::{Serialize, Deserialize};
use crate::meta::determinism::hash_bytes;

#[derive(Clone, Debug, Serialize, Deserialize)]
pub struct Block {
    pub input_hash: [u8; 32],
    pub output_hash: [u8; 32],
    pub prev_block_hash: [u8; 32],
    pub block_hash: [u8; 32],
    pub ok: bool,
}

impl Block {
    pub fn new(input: &[u8], output: &[u8], prev: [u8;32], ok: bool) -> Self {
        let input_hash = hash_bytes(input);
        let output_hash = hash_bytes(output);
        let mut buf = Vec::new();
        buf.extend_from_slice(&input_hash);
        buf.extend_from_slice(&output_hash);
        buf.extend_from_slice(&prev);
        let block_hash = hash_bytes(&buf);
        Self { input_hash, output_hash, prev_block_hash: prev, block_hash, ok }
    }
    pub fn validate(&self) -> bool {
        let mut buf = Vec::new();
        buf.extend_from_slice(&self.input_hash);
        buf.extend_from_slice(&self.output_hash);
        buf.extend_from_slice(&self.prev_block_hash);
        hash_bytes(&buf) == self.block_hash
    }
}

System – chain.rs

use super::block::Block;

#[derive(Debug)]
pub struct Chain {
    pub blocks: Vec<Block>,
    pub genesis_hash: [u8; 32],
}

impl Chain {
    pub fn new(genesis_hash: [u8; 32]) -> Self {
        Self { blocks: Vec::new(), genesis_hash }
    }
    pub fn last_hash(&self) -> [u8; 32] {
        self.blocks.last().map(|b| b.block_hash).unwrap_or(self.genesis_hash)
    }
    pub fn append(&mut self, block: Block) -> Result<(), String> {
        if block.prev_block_hash != self.last_hash() {
            return Err("prev_block_hash mismatch".into());
        }
        if !block.validate() {
            return Err("block hash invalid".into());
        }
        self.blocks.push(block);
        Ok(())
    }
    pub fn validate_chain(&self) -> bool {
        let mut prev = self.genesis_hash;
        for b in &self.blocks {
            if b.prev_block_hash != prev || !b.validate() {
                return false;
            }
            prev = b.block_hash;
        }
        true
    }
}

Consensus – commit.rs

use crate::system::node::Node;

pub fn try_commit(node: &mut Node, state: &mut ConsensusState, quorum: usize) -> Result<(), String> {
    if state.pending_block.is_none() {
        return Err("no pending block".into());
    }
    if state.accepts.len() < quorum {
        return Err("no quorum".into());
    }
    let block = state.pending_block.take().unwrap();
    node.append_block(block)?;
    state.accepts.clear();
    Ok(())
}

Identity – identity.rs

use crate::meta::determinism::hash_bytes;

pub struct Identity {
    pub global_state_hash: [u8; 32],
    pub graph_digest_hash: [u8; 32],
    pub identity_hash: [u8; 32],
}

impl Identity {
    pub fn compute(global_state_hash: [u8; 32], graph_digest_hash: [u8; 32]) -> Self {
        let mut buf = Vec::with_capacity(64);
        buf.extend_from_slice(&global_state_hash);
        buf.extend_from_slice(&graph_digest_hash);
        let identity_hash = hash_bytes(&buf);
        Self { global_state_hash, graph_digest_hash, identity_hash }
    }
}
